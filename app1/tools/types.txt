# --- time stretch / pitch shift (docs/app1/topics/time-stretch.md) ---
struct StretchRing 0xA060           # source ring of one channel; header + 0x2800 samples
  0x000  int[5] block_pos           # source position of each 0x800-sample block (returned via pos_out)
  0x014  uint   write_idx           # filled up to here by Voice_RenderSetup
  0x018  uint   read_block          # read_idx >> 7, set before the guard/fallback helpers run
  0x024  uint   ideal_pos           # (left ring only) where the read should be: Q14, wraps at 0x2800 << 14
  0x028  uint   corr_anchor         # (left ring only) start of the Ensemble correlation window (read + 1200)
  0x038  uint   read_idx            # integer read position; Ring_Advance moves it
  0x03C  float  elapsed             # (left ring only) running sum of ratio + modulation
  0x040  float  mod_phase           # (left ring only) position inside the modulation cycle
  0x044  float  elapsed2            # (left ring only) compared with StretchState.mod_switch_at
  0x048  int    moved               # total of all Ring_Advance steps
  0x060  int[10240] data            # int16 << 13 (full scale +-0x10000000)
end

struct StretchState 0x328           # one per voice at g_stretch_state (OCRAM 0x2023F000 + i*0x328)
  0x000  uint   frac                # fractional read position, Q14
  0x008  int    holdoff             # samples until the next splice is allowed (0x400 after a splice; guard adds 2000)
  0x010  int    xfade_left          # samples left in the current crossfade
  0x018  int    xfade_step          # Q28 gain step = 0x10000000 / (period - 1)
  0x020  uint   guard_env           # Stretch_TransientGuard: peak level, decays x0.9999 per sample
  0x02C  int    guard_hold          # Stretch_TransientGuard: re-arm delay (g_guard_hold)
  0x030  int    guard_count         # Stretch_TransientGuard: samples since the level last rose (-1 = armed)
  0x034  int    calls               # Resampler_Process calls
  0x048  byte   splice              # 0 straight read, 1 crossfading forward (skip), 2 crossfading back (repeat)
  0x04C  int    corr_k              # Ensemble: lag index 0..100 (lag = 400 + 5k); -1 before the first run
  0x050  short  corr_done
  0x054  int    period              # splice length in samples (Off 600, Backing 400, Ensemble 800 at start)
  0x058  int    best_lag            # Ensemble: lag with the highest correlation so far
  0x05C  int    detected_period     # Ensemble: best_lag after a full sweep (-1 = none yet)
  0x068  int    best_corr
  0x06C  int    corr_norm           # energy of the window (used only when 0x805968DD is set)
  0x2A0  int    gain_out            # Q28 gain of the tap being faded out
  0x2A4  int    gain_in             # Q28 gain of the tap being faded in
  0x2A8  byte   jump_periods        # 1 or 2: how many periods a splice skips/repeats
  0x2AC  int    corr_last
  0x2B0  int    corr_last_norm
  0x2B8  int    mod_len             # ratio modulation: cycle length in samples (from the sample BPM)
  0x2BC  float  mod_q1              # mod_len * 0.25
  0x2C0  float  mod_q2              # mod_len * 0.5
  0x2C4  float  mod_q3              # mod_len * 0.75
  0x2C8  float  mod_depth           # Stretch_ClampRatio keeps ratio +- this inside 0.1..1.9
  0x2CC  float  mod_depth_base
  0x2D0  int    mod_shape           # pad param 0x8A; 0 = no modulation, shapes 1..8 ((n-1) % 4 picks the curve)
  0x2D8  double mod_slope
  0x2E0  double mod_value
  0x2E8  int    mod_random          # pad param 0x8C: 1/2/3 = +-7/14/21 % random depth per cycle (Rand_Next)
  0x2EC  byte   mod_pending         # new modulation settings waiting at 0x2F0..0x320
  0x324  int    mod_switch_at
end
